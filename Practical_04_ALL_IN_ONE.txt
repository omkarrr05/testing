PRACTICAL No. 04 - Data Extraction and Score Analysis from Web Table
======================================================================

TOOLS USED
----------
  * Selenium WebDriver (Java)
  * Selenium JAR + ChromeDriver
  * NetBeans 8.2 + JDK 1.8
  * XAMPP (Apache, MySQL, PHP) + phpMyAdmin (dynamic table)

REQUIREMENTS
------------
  * HTML form : YES - student.html (static table) and student.php (dynamic table from MySQL)
  * Excel data : NOT needed (data lives in HTML / MySQL)
  * Java code  : YES - WebTableHandling.java (static) and CountStudents.java (dynamic)
  * Database   : YES - student_marks.students (student_marks.sql)

FILES IN THIS PRACTICAL
-----------------------
  - student.html
  - student.php
  - student_marks.sql
  - WebTableHandling.java
  - CountStudents.java

STEPS / HOW TO RUN
------------------
PART 1 - static table
1. Put student.html in C:\xampp\htdocs\project4\
2. Run WebTableHandling.java: prints the header row and all 8 student rows with printf("%-11s").
PART 2 - dynamic table
3. phpMyAdmin -> run student_marks.sql (creates database student_marks, table students, 8 rows).
4. Put student.php in C:\xampp\htdocs\project4\ ; the PHP connects on port 3308 -> change to 3306 if your MySQL uses the default.
5. Run CountStudents.java (page: http://localhost:8080/project4/student.php).
   It prints: total students, students scoring above 60 in at least one subject, pass / fail count and percentages
   (fail = any subject mark below 40).
Edit chromedriver.exe paths and the Apache port in the Java files for your machine.

package Project4;

import java.util.List;
import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;

public class WebTableHandling {

    public static void main(String[] args) {

        System.setProperty("webdriver.chrome.driver",
                "C:\\Users\\Sakshi\\Documents\\NetBeansProjects\\project4\\src\\driver\\chromedriver.exe");

        WebDriver driver = new ChromeDriver();

        driver.get("file:///C:/xampp/htdocs/project4/student.html");

        WebElement table = driver.findElement(By.tagName("table"));

        List<WebElement> rows = table.findElements(By.tagName("tr"));

        for (WebElement row : rows) {

            List<WebElement> cells = row.findElements(By.tagName("th"));

            if (cells.isEmpty()) {
                cells = row.findElements(By.tagName("td"));
            }

            for (WebElement cell : cells) {
                System.out.printf("%-11s", cell.getText());
            }

            System.out.println();
        }
    }
}

package CountingStudents;

import java.util.List;
import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;

public class CountStudents {

    public static void main(String[] args) {

        System.setProperty(
                "webdriver.chrome.driver",
                "C:\\Users\\Sakshi\\Documents\\NetBeansProjects\\project4\\src\\driver\\chromedriver.exe");

        WebDriver driver = new ChromeDriver();

        try {
            driver.get("http://localhost:8080/project4/student.php");

            List<WebElement> rows = driver.findElements(By.xpath("//table//tr"));

            int totalStudents = rows.size() - 1;
            int matchingStudentCount = 0;
            int passCount = 0;
            int failCount = 0;

            for (int i = 1; i < rows.size(); i++) {

                WebElement row = rows.get(i);
                List<WebElement> cells = row.findElements(By.tagName("td"));

                boolean scoredAbove60 = false;
                boolean pass = true;

                for (WebElement cell : cells) {

                    String cellText = cell.getText().trim();

                    if (!cellText.isEmpty()) {
                        try {
                            int mark = Integer.parseInt(cellText);

                            // Student scored above 60 in at least one subject
                            if (mark > 60) {
                                scoredAbove60 = true;
                            }

                            // Student fails if any subject is below 40
                            if (mark < 40) {
                                pass = false;
                            }
                        } catch (NumberFormatException e) {
                            // Ignore non-numeric cells (the student name)
                        }
                    }
                }

                if (scoredAbove60) {
                    matchingStudentCount++;
                }

                if (pass) {
                    passCount++;
                } else {
                    failCount++;
                }
            }

            double percentage = 0.0;
            if (totalStudents > 0) {
                percentage = ((double) matchingStudentCount / totalStudents) * 100;
            }

            double passPercentage = 0.0;
            if (totalStudents > 0) {
                passPercentage = ((double) passCount / totalStudents) * 100;
            }

            System.out.println("Total Students Processed : " + totalStudents);
            System.out.println("Students Scoring Above 60 in At Least One Subject : " + matchingStudentCount);
            System.out.printf("Percentage Above 60 : %.2f%%%n", percentage);
            System.out.println("Pass Students : " + passCount);
            System.out.println("Fail Students : " + failCount);
            System.out.printf("Pass Percentage : %.2f%%%n", passPercentage);

        } finally {
        }
    }
}
